%
%   acmart                          ieeeconf
%   ------------------------------  --------------------------------------------
%   \documentclass[sigconf,...]     \documentclass[letterpaper,10pt,conference]
%   \settopmatter / \setcopyright   dropped; ieeeconf has no ACM front matter
%   \citestyle{acmnumeric}          dropped; IEEEtran.bst is numeric already
%   \bibliographystyle{plainnat}    \bibliographystyle{IEEEtran}
%   \title[short]{long}             \title{...}; ieeeconf has no running short title
%   \keywords{}                     dropped; ICRA takes keywords through PaperPlaza
%   \Description{...}               commented out in the section files; acmart-only
%   graphicx loaded by the class    loaded explicitly here
%
% ICRA is not double-blind, so the anonymous option is gone and the author block is
% filled in below.
