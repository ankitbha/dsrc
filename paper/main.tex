% ICRA / IEEE conference format.
%
% Ported from the earlier ACM `acmart[sigconf]` submission, whose source this shares.
% What changed and why, so the next person does not have to diff two classes:
%
%   acmart                          ieeeconf
%   ------------------------------  --------------------------------------------
%   \documentclass[sigconf,...]     \documentclass[letterpaper,10pt,conference]
%   \settopmatter / \setcopyright   dropped; ieeeconf has no ACM front matter
%   \citestyle{acmnumeric}          dropped; IEEEtran.bst is numeric already
%   \bibliographystyle{plainnat}    \bibliographystyle{IEEEtran}
%   \title[short]{long}             \title{...}; ieeeconf has no running short title
%   \keywords{}                     dropped; ICRA takes keywords through PaperPlaza
%   \Description{...}               commented out in the section files; acmart-only
%   graphicx loaded by the class    loaded explicitly here
%
% ICRA is double-blind, so the author block stays anonymous until the camera-ready.
% Two things follow from that and are easy to undo by accident. The affiliation and
% e-mail in \author are placeholders, and prior work by these authors is cited in the
% third person, never as "we": see the Mainz paragraph in intro.tex.
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\documentclass[letterpaper, 10pt, conference]{ieeeconf}

\IEEEoverridecommandlockouts   % needed for \thanks
\overrideIEEEmargins           % meets the printer requirements

\usepackage{graphicx}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{array}
\usepackage{booktabs}
% enumitem clashes with ieeeconf, which already defines \labelindent. Freeing the name
% before loading it is the usual fix; no section uses an enumitem option list, so it is
% loaded only because the ACM source loaded it.
\makeatletter\let\labelindent\relax\makeatother
\usepackage{enumitem}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,calc,positioning,fit}

% Float parameters. The defaults reserve so much of each column for text that a
% half-column figure is deferred to the end of the document rather than placed;
% all three figures here did exactly that before these were raised.
\renewcommand{\topfraction}{0.88}
\renewcommand{\bottomfraction}{0.7}
\renewcommand{\textfraction}{0.1}
\renewcommand{\floatpagefraction}{0.75}
\renewcommand{\dbltopfraction}{0.88}
\renewcommand{\dblfloatpagefraction}{0.75}
\setcounter{topnumber}{3}
\setcounter{dbltopnumber}{3}
\setcounter{totalnumber}{4}

\title{\LARGE \bf Realizing Self-Regulating Cars on Commodity Hardware}

\author{Anonymous Authors$^{1}$%
\thanks{$^{1}$Anonymous Affiliations. {\tt\small anonymous.emails}}%
}

\maketitle
\thispagestyle{empty}

\input{discussion}

\end{document}
